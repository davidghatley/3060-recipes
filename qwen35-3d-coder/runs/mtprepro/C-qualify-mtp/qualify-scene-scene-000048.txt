    // ---------------------------------------------------------------------
    // Custom orbit controls
    // ---------------------------------------------------------------------

    const canvas = renderer.domElement;
    const pointers = new Map();
    let lastPinchDistance = 0;
    let dragging = false;
    let lastInteraction = performance.now();

    function pointerDistance() {
      const values = [...pointers.values()];
      if (values.length < 2) return 0;

      return Math.hypot(
        values[0].x - values[1].x,
        values[0].y - values[1].y
      );
    }

    canvas.addEventListener("pointerdown", (event) => {
      pointers.set(event.pointerId, {
        x: event.clientX,
        y: event.clientY
      });

      canvas.setPointerCapture(event.pointerId);
      dragging = true;
      lastInteraction = performance.now();

      if (pointers.size === 2) {
        lastPinchDistance = pointerDistance();
      }
    });

    canvas.addEventListener("pointermove", (event) => {
      const previous = pointers.get(event.pointerId);
      if (!previous) return;

      const dx = event.clientX - previous.x;
      const dy = event.clientY - previous.y;

      pointers.set(event.pointerId, {
        x: event.clientX,
        y: event.clientY
      });

      if (pointers.size === 1) {
        desiredView.theta -= dx * 0.006;
        desiredView.phi = THREE.MathUtils.clamp(
          desiredView.phi - dy * 0.005,
          0.35,
          1.48
        );
      } else if (pointers.size === 2) {
        const distance = pointerDistance();

        if (lastPinchDistance > 0 && distance > 0) {
          desiredView.radius = THREE.MathUtils.clamp(
            desiredView.radius * (lastPinchDistance / distance),
            5.4,
            15
          );
        }

        lastPinchDistance = distance;
      }

      lastInteraction = performance.now();
    });

    function endPointer(event) {
      pointers.delete(event.pointerId);
      dragging = pointers.size > 0;
      lastPinchDistance = 0;
      lastInteraction = performance.now();
    }

    canvas.addEventListener("pointerup", endPointer);
    canvas.addEventListener("pointercancel", endPointer);
    canvas.addEventListener("lostpointercapture", endPointer);

    canvas.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        desiredView.radius = THREE.MathUtils.clamp(
          desiredView.radius * Math.exp(event.deltaY * 0.001),
          5.4,
          15
        );
        lastInteraction = performance.now();
      },
      { passive: false }
    );

    function resetView() {
      desiredView.theta = initialView.theta;
      desiredView.phi = initialView.phi;
      desiredView.radius = initialView.radius;
      lastInteraction = performance.now();
    }

    canvas.addEventListener("dblclick", resetView);

    let windOn = true;

    // ---------------------------------------------------------------------
    // Animation
    // ---------------------------------------------------------------------

    const clock = new THREE.Clock();
    const eyePosition = new THREE.Vector3();
    const reducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;

    function animate() {
      const delta = Math.min(clock.getDelta(), 0.05);
      const elapsed = clock.elapsedTime;

      // Smooth camera movement.
      const cameraEase = 1 - Math.exp(-delta * 8);

      currentView.theta +=
        (desiredView.theta - currentView.theta) * cameraEase;
      currentView.phi +=
        (desiredView.phi - currentView.phi) * cameraEase;
      currentView.radius +=
        (desiredView.radius - currentView.radius) * cameraEase;

      if (!dragging && !reducedMotion) {
        desiredView.theta += delta * 0.075;
      }

      const sinPhi = Math.sin(currentView.phi);

      camera.position.set(
        cameraTarget.x +
          currentView.radius * sinPhi * Math.sin(currentView.theta),
        cameraTarget.y + currentView.radius * Math.cos(currentView.phi),
        cameraTarget.z +
          currentView.radius * sinPhi * Math.cos(currentView.theta)
      );

      camera.lookAt(cameraTarget);

      // Gentle kite motion driven by the wind.
      if (windOn) {
        kite.position.x = Math.sin(elapsed * 0.72) * 0.13;
        kite.position.y = 0.12 + Math.sin(elapsed * 0.58) * 0.075;
        kite.rotation.z = Math.sin(elapsed * 0.64) * 0.025;
        kite.rotation.x = Math.sin(elapsed * 0.48) * 0.014;
        kite.rotation.y = Math.sin(elapsed * 0.36) * 0.035;
      } else {
        kite.position.set(0, 0.12, 0);
        kite.rotation.set(0, 0, 0);
      }

      kite.updateMatrixWorld(true);
      eyePosition.set(0, 1.93, 0).applyMatrix4(kite.matrixWorld);

      stringPositions.setXYZ(
        0,
        anchor.x,
        anchor.y,
        anchor.z
      );
      stringPositions.setXYZ(
        3,
        eyePosition.x,
        eyePosition.y,
        eyePosition.z
      );
      stringPositions.needsUpdate = true;

      // Drifting wind particles.
      if (windOn) {
        const positions = particleGeometry.attributes.position.array;

        for (let i = 0; i < particleCount; i++) {
          const particle = particleData[i];
          positions[i * 3] += particle.speed * delta;

          if (positions[i * 3] > 11) {
            positions[i * 3] = -11;
          }

          positions[i * 3 + 1] +=
            Math.sin(elapsed * 0.8 + particle.phase) * delta * 0.025;
        }

        particleGeometry.attributes.position.needsUpdate = true;
      }

      renderer.render(scene, camera);
    }

    function onResize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();

      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    }

    window.addEventListener("resize", onResize);

    renderer.setAnimationLoop(animate);

    requestAnimationFrame(() => {
      loading.classList.add("hidden");
    });

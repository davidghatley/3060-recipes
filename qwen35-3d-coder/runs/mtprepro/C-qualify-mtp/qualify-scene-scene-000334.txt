    function createSpongeTextures() {
      const size = 1024;
      const colorCanvas = document.createElement("canvas");
      const bumpCanvas = document.createElement("canvas");

      colorCanvas.width = colorCanvas.height = size;
      bumpCanvas.width = bumpCanvas.height = size;

      const colorContext = colorCanvas.getContext("2d");
      const bumpContext = bumpCanvas.getContext("2d");

      const baseGradient = colorContext.createRadialGradient(
        size * 0.48,
        size * 0.42,
        size * 0.08,
        size * 0.5,
        size * 0.5,
        size * 0.72
      );

      baseGradient.addColorStop(0, "#e8c992");
      baseGradient.addColorStop(0.45, "#d6b981");
      baseGradient.addColorStop(1, "#b99762");

      colorContext.fillStyle = baseGradient;
      colorContext.fillRect(0, 0, size, size);

      bumpContext.fillStyle = "#808080";
      bumpContext.fillRect(0, 0, size, size);

      // Draw hundreds of irregular pores.
      for (let i = 0; i < 2100; i++) {
        const x = Math.random() * size;
        const y = Math.random() * size;
        const radius = 0.7 + Math.random() * 6.5;
        const darkness = 0.12 + Math.random() * 0.32;

        const colorGradient = colorContext.createRadialGradient(
          x,
          y,
          0,
          x,
          y,
          radius
        );

        colorGradient.addColorStop(
          0,
          `rgba(73, 43, 24, ${darkness})`
        );
        colorGradient.addColorStop(1, "rgba(73, 43, 24, 0)");

        colorContext.fillStyle = colorGradient;
        colorContext.beginPath();
        colorContext.arc(x, y, radius, 0, Math.PI * 2);
        colorContext.fill();

        const bumpGradient = bumpContext.createRadialGradient(
          x,
          y,
          0,
          x,
          y,
          radius
        );

        bumpGradient.addColorStop(0, "rgba(255,255,255,0.7)");
        bumpGradient.addColorStop(0.5, "rgba(45,45,45,0.45)");
        bumpGradient.addColorStop(1, "rgba(45,45,45,0)");

        bumpContext.fillStyle = bumpGradient;
        bumpContext.beginPath();
        bumpContext.arc(x, y, radius, 0, Math.PI * 2);
        bumpContext.fill();
      }

      // Fine sandy texture.
      for (let i = 0; i < 5000; i++) {
        const x = Math.random() * size;
        const y = Math.random() * size;
        const alpha = 0.025 + Math.random() * 0.065;

        colorContext.fillStyle = `rgba(255,244,210,${alpha})`;
        colorContext.fillRect(x, y, 1 + Math.random() * 2, 1 + Math.random() * 2);

        const value = 115 + Math.random() * 100;
        bumpContext.fillStyle = `rgba(${value},${value},${value},0.12)`;
        bumpContext.fillRect(x, y, 1, 1);
      }

      const colorTexture = new THREE.CanvasTexture(colorCanvas);
      colorTexture.colorSpace = THREE.SRGBColorSpace;
      colorTexture.wrapS = colorTexture.wrapT = THREE.RepeatWrapping;
      colorTexture.repeat.set(2.3, 1.45);
      colorTexture.anisotropy = renderer.capabilities.getMaxAnisotropy();

      const bumpTexture = new THREE.CanvasTexture(bumpCanvas);
      bumpTexture.wrapS = bumpTexture.wrapT = THREE.RepeatWrapping;
      bumpTexture.repeat.copy(colorTexture.repeat);
      bumpTexture.anisotropy = colorTexture.anisotropy;

      return { colorTexture, bumpTexture };
    }

    function createCausticTexture() {
      const canvas = document.createElement("canvas");
      canvas.width = canvas.height = 512;

      const context = canvas.getContext("2d");
      const gradient = context.createRadialGradient(
        256,
        256,
        20,
        256,
        256,
        360
      );

      gradient.addColorStop(0, "rgba(124,220,210,0.18)");
      gradient.addColorStop(0.5, "rgba(70,153,157,0.08)");
      gradient.addColorStop(1, "rgba(0,0,0,0)");

      context.fillStyle = gradient;
      context.fillRect(0, 0, 512, 512);

      for (let i = 0; i < 90; i++) {
        const x = Math.random() * 512;
        const y = Math.random() * 512;
        const radiusX = 25 + Math.random() * 115;
        const radiusY = 15 + Math.random() * 65;
        const rotation = Math.random() * Math.PI;

        context.beginPath();
        context.ellipse(
          x,
          y,
          radiusX,
          radiusY,
          rotation,
          0,
          Math.PI * 2
        );

        context.strokeStyle = `rgba(153,235,221,${
          0.035 + Math.random() * 0.075
        })`;
        context.lineWidth = 0.5 + Math.random() * 1.8;
        context.stroke();
      }

      const texture = new THREE.CanvasTexture(canvas);
      texture.colorSpace = THREE.SRGBColorSpace;
      texture.wrapS = texture.wrapT = THREE.RepeatWrapping;
      texture.repeat.set(5, 5);
      texture.anisotropy = renderer.capabilities.getMaxAnisotropy();

      return texture;
    }

    function createTaperedTube(
      curve,
      radiusStart,
      radiusEnd,
      longitudinalSegments = 84,
      radialSegments = 10
    ) {
      const frames = curve.computeFrenetFrames(
        longitudinalSegments,
        false
      );

      const positions = [];
      const normals = [];
      const uvs = [];
      const indices = [];

      for (let i = 0; i <= longitudinalSegments; i++) {
        const t = i / longitudinalSegments;
        const center = curve.getPointAt(t);
        const normal = frames.normals[i];
        const binormal = frames.binormals[i];

        const radius = THREE.MathUtils.lerp(
          radiusStart,
          radiusEnd,
          Math.pow(t, 0.72)
        );

        for (let j = 0; j <= radialSegments; j++) {
          const v = (j / radialSegments) * Math.PI * 2;
          const cosine = Math.cos(v);
          const sine = Math.sin(v);

          const radialNormal = new THREE.Vector3()
            .copy(normal)
            .multiplyScalar(cosine)
            .addScaledVector(binormal, sine)
            .normalize();

          const vertex = center.clone().addScaledVector(
            radialNormal,
            radius
          );

          positions.push(vertex.x, vertex.y, vertex.z);
          normals.push(
            radialNormal.x,
            radialNormal.y,
            radialNormal.z
          );

          uvs.push(j / radialSegments, t);
        }
      }

      for (let i = 0; i < longitudinalSegments; i++) {
        for (let j = 0; j < radialSegments; j++) {
          const a = i * (radialSegments + 1) + j;
          const b = (i + 1) * (radialSegments + 1) + j;
          const c = (i + 1) * (radialSegments + 1) + j + 1;
          const d = i * (radialSegments + 1) + j + 1;

          indices.push(a, b, d, b, c, d);
        }
      }

      const geometry = new THREE.BufferGeometry();
      geometry.setAttribute(
        "position",
        new THREE.Float32BufferAttribute(positions, 3)
      );
      geometry.setAttribute(
        "normal",
        new THREE.Float32BufferAttribute(normals, 3)
      );
      geometry.setAttribute(
        "uv",
        new THREE.Float32BufferAttribute(uvs, 2)
      );
      geometry.setIndex(indices);
      geometry.computeBoundingSphere();

      return geometry;
    }

    function updateCamera(delta) {
      const smoothing = 1 - Math.exp(-delta * 8);

      current.radius = THREE.MathUtils.lerp(
        current.radius,
        desired.radius,
        smoothing
      );

      current.theta = THREE.MathUtils.lerp(
        current.theta,
        desired.theta,
        smoothing
      );

      current.phi = THREE.MathUtils.lerp(
        current.phi,
        desired.phi,
        smoothing
      );

      const sinPhi = Math.sin(current.phi);

      camera.position.set(
        target.x +
          current.radius * sinPhi * Math.sin(current.theta),

        target.y + current.radius * Math.cos(current.phi),

        target.z +
          current.radius * sinPhi * Math.cos(current.theta)
      );

      camera.lookAt(target);
    }

    function resetView() {
      desired.radius = defaultView.radius;
      desired.theta = defaultView.theta;
      desired.phi = defaultView.phi;
    }

    renderer.domElement.addEventListener("pointerdown", (event) => {
      dragging = true;
      previousX = event.clientX;
      previousY = event.clientY;
      renderer.domElement.setPointerCapture(event.pointerId);
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - previousX;
      const deltaY = event.clientY - previousY;

      previousX = event.clientX;
      previousY = event.clientY;

      desired.theta -= deltaX * 0.006;
      desired.phi = THREE.MathUtils.clamp(
        desired.phi - deltaY * 0.005,
        0.32,
        Math.PI * 0.485
      );
    });

    function endDrag(event) {
      dragging = false;

      if (
        event.pointerId !== undefined &&
        renderer.domElement.hasPointerCapture(event.pointerId)
      ) {
        renderer.domElement.releasePointerCapture(event.pointerId);
      }
    }

    renderer.domElement.addEventListener("pointerup", endDrag);
    renderer.domElement.addEventListener("pointercancel", endDrag);

      if (firstFrame) {

      }

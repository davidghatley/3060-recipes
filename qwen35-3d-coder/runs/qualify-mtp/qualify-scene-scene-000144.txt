    function createSolarTexture() {
      const canvas = document.createElement("canvas");
      canvas.width = 1024;
      canvas.height = 640;
      const ctx = canvas.getContext("2d");

      const columns = 12;
      const rows = 8;
      const cellWidth = canvas.width / columns;
      const cellHeight = canvas.height / rows;

      let seed = 9137;
      const random = () => {
        seed = (seed * 16807) % 2147483647;
        return (seed - 1) / 2147483646;
      };

      for (let row = 0; row < rows; row++) {
        for (let column = 0; column < columns; column++) {
          const x = column * cellWidth;
          const y = row * cellHeight;
          const w = cellWidth - 5;
          const h = cellHeight - 5;

          const gradient = ctx.createLinearGradient(x, y, x + w, y + h);
          const hue = 215 + Math.floor(random() * 18);
          gradient.addColorStop(0, `hsl(${hue}, 72%, 22%)`);
          gradient.addColorStop(0.5, `hsl(${hue}, 76%, 17%)`);
          gradient.addColorStop(1, `hsl(${hue}, 68%, 14%)`);

          ctx.fillStyle = gradient;
          ctx.fillRect(x + 2, y + 2, w, h);

          ctx.strokeStyle = "rgba(125, 195, 255, 0.28)";
          ctx.lineWidth = 1.2;
          ctx.strokeRect(x + 2, y + 2, w, h);

          // Fine busbars across each cell.
          ctx.strokeStyle = "rgba(155, 216, 255, 0.38)";
          ctx.lineWidth = 0.75;
          for (let bus = 0; bus < 4; bus++) {
            const bx = x + 5 + bus * (w - 10) / 3;
            ctx.beginPath();
            ctx.moveTo(bx, y + 3);
            ctx.lineTo(bx, y + h - 3);
            ctx.stroke();
          }
        }
      }

      // Subtle manufacturing texture.
      for (let i = 0; i < 2600; i++) {
        const x = random() * canvas.width;
        const y = random() * canvas.height;
        const alpha = 0.015 + random() * 0.035;
        ctx.fillStyle = random() > 0.5
          ? `rgba(255,255,255,${alpha})`
          : `rgba(0,0,0,${alpha})`;
        ctx.fillRect(x, y, 1 + random() * 2, 1 + random() * 2);
      }

      const texture = new THREE.CanvasTexture(canvas);
      texture.colorSpace = THREE.SRGBColorSpace;
      texture.anisotropy = renderer.capabilities.getMaxAnisotropy();
      return texture;
    }

    function createConcreteTexture() {
      const canvas = document.createElement("canvas");
      canvas.width = 512;
      canvas.height = 512;
      const ctx = canvas.getContext("2d");
      const image = ctx.createImageData(512, 512);

      let seed = 441;
      const random = () => {
        seed = (seed * 48271) % 2147483647;
        return (seed - 1) / 2147483646;
      };

      for (let i = 0; i < image.data.length; i += 4) {
        const value = 153 + Math.floor(random() * 28);
        image.data[i] = value;
        image.data[i + 1] = value - 2;
        image.data[i + 2] = value - 7;
        image.data[i + 3] = 255;
      }

      ctx.putImageData(image, 0, 0);
      ctx.strokeStyle = "rgba(65,65,62,0.18)";
      ctx.lineWidth = 2;

      for (let i = 0; i < 8; i++) {
        const y = i * 64;
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(512, y + Math.sin(i * 2.1) * 4);
        ctx.stroke();
      }

      const texture = new THREE.CanvasTexture(canvas);
      texture.colorSpace = THREE.SRGBColorSpace;
      texture.wrapS = THREE.RepeatWrapping;
      texture.wrapT = THREE.RepeatWrapping;
      texture.repeat.set(7, 7);
      texture.anisotropy = renderer.capabilities.getMaxAnisotropy();
      return texture;
    }

    function addCylinderBetween(
      start,
      end,
      radius,
      material,
      parent = scene,
      radialSegments = 16

      const a = new THREE.Vector3(...start);
      const b = new THREE.Vector3(...end);
      const direction = new THREE.Vector3().subVectors(b, a);

      const mesh = new THREE.Mesh(
        new THREE.CylinderGeometry(radius, radius, length, radialSegments),
        material

        new THREE.Vector3(0, 1, 0),
        direction.normalize()
      );
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      parent.add(mesh);
      return mesh;

            knobCenter.y - knobAxis.y * 0.09,
            knobCenter.z - knobAxis.z * 0.09
          ],
          [
            knobCenter.x + knobAxis.x * 0.09,
            knobCenter.y + knobAxis.y * 0.09,
            knobCenter.z + knobAxis.z * 0.09
          ],
          0.125,
          materials.darkSteel,
          scene,
          20
        );

        const wing = new THREE.Mesh(
          new THREE.TorusGeometry(0.115, 0.022, 8, 20),
          materials.silver
        );
        wing.position.copy(knobCenter).addScaledVector(knobAxis, 0.105);
        wing.quaternion.setFromUnitVectors(
          new THREE.Vector3(0, 0, 1),
          knobAxis
        );
        wing.castShadow = true;
        scene.add(wing);

        // Small saddle where the strut meets the rail.
        addBox(
          [0.18, 0.14, 0.22],
          materials.darkSteel,
          [lowerJoint[0], lowerJoint[1], lowerJoint[2]]
        );
      }
    }

    // ---------------------------------------------------------------------
    // Photovoltaic module
    // ---------------------------------------------------------------------
    const module = new THREE.Group();
    module.position.set(0, 2.66, 0);
    module.rotation.x = 0.38;
    scene.add(module);

    // Dark blue photovoltaic cell field.
    const solarField = new THREE.Mesh(
      new THREE.PlaneGeometry(3.84, 2.08),
      materials.solar
    );
    solarField.position.z = 0.008;
    solarField.receiveShadow = true;
    module.add(solarField);

    // Thin silver perimeter frame.
    addBox(
      [3.96, 0.08, 0.14],
      materials.silver,
      [0, 1.11, 0],
      module
    );
    addBox(
      [3.96, 0.08, 0.14],
      materials.silver,
      [0, -1.11, 0],
      module
    );
    addBox(
      [0.08, 2.14, 0.14],
      materials.silver,
      [-1.91, 0, 0],
      module
    );
    addBox(
      [0.08, 2.14, 0.14],
      materials.silver,
      [1.91, 0, 0],
      module
    );

    // Frame fasteners.
    for (const x of [-1.82, 1.82]) {
      for (const y of [-1.01, 1.01]) {
        addScrew([x, y, 0.092], module);
      }
    }

    // Junction box on the back of the module.
    addBox(
      [0.58, 0.4, 0.16],
      materials.black,
      [0.65, -0.5, -0.16],
      module

    addBox(
      [0.62, 0.05, 0.2],
      materials.black,
      [0.65, 0.225, -0.16],

    for (const dx of [-0.24, 0.24]) {
      for (const dy of [-0.13, 0.13]) {
        addScrew([0.65 + dx, 0.255, -0.055], module, 0.035);

    const label = new THREE.Mesh(
      new THREE.PlaneGeometry(0.34, 0.13),

    label.rotation.x = -Math.PI / 2;
    label.position.set(0.65, 0.29, -0.055);
    module.add(label);

    // Cable running from the junction box to ground.
    module.updateMatrixWorld(true);

    const boxWorld = module.localToWorld(new THREE.Vector3(0.65, -0.5, -0.24));
    const cableCurve = new THREE.CatmullRomCurve3([

      boxWorld.clone().add(new THREE.Vector3(0.02, -0.23, 0)),

    addCylinderBetween(

    // ---------------------------------------------------------------------
    // Custom camera controls: orbit, zoom, keyboard, and touch pinch
    // ---------------------------------------------------------------------
    const controls = {
      target: new THREE.Vector3(0, 1.72, 0),
      radius: 9.4,
      theta: 2.42,
      phi: 1.08
    };

    const initialControls = {
      radius: controls.radius,
      theta: controls.theta,
      phi: controls.phi
    };

    function updateCamera() {
      const sinPhi = Math.sin(controls.phi);

      camera.position.set(
        controls.target.x +
          controls.radius * sinPhi * Math.sin(controls.theta),
        controls.target.y + controls.radius * Math.cos(controls.phi),
        controls.target.z +
          controls.radius * sinPhi * Math.cos(controls.theta)
      );

      camera.lookAt(controls.target);
    }

    function resetView() {
      controls.radius = initialControls.radius;
      controls.theta = initialControls.theta;
      controls.phi = initialControls.phi;
      updateCamera();
    }

    const pointers = new Map();
    let lastX = 0;
    let lastY = 0;
    let lastPinchDistance = 0;

    function pointerDistance() {
      const values = [...pointers.values()];
      if (values.length < 2) return 0;
      return Math.hypot(
        values[0].x - values[1].x,
        values[0].y - values[1].y
      );
    }

    renderer.domElement.addEventListener("pointerdown", (event) => {
      renderer.domElement.setPointerCapture(event.pointerId);
      pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });

      if (pointers.size === 1) {
        lastX = event.clientX;
        lastY = event.clientY;
      } else if (pointers.size === 2) {
        lastPinchDistance = pointerDistance();
      }
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!pointers.has(event.pointerId)) return;

      pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });

      if (pointers.size === 1) {
        const dx = event.clientX - lastX;
        const dy = event.clientY - lastY;
        lastX = event.clientX;
        lastY = event.clientY;

        controls.theta -= dx * 0.006;
        controls.phi = THREE.MathUtils.clamp(
          controls.phi - dy * 0.005,
          0.32,
          1.48
        );
      } else if (pointers.size === 2) {
        const distance = pointerDistance();
        if (lastPinchDistance > 0 && distance > 0) {
          controls.radius = THREE.MathUtils.clamp(
            controls.radius * (lastPinchDistance / distance),
            5.2,
            16
          );
        }
        lastPinchDistance = distance;
      }

      updateCamera();
    });

    function endPointer(event) {
      pointers.delete(event.pointerId);

      if (pointers.size === 1) {
        const remaining = [...pointers.values()][0];
        lastX = remaining.x;
        lastY = remaining.y;
      }

      lastPinchDistance = pointers.size === 2 ? pointerDistance() : 0;
    }

    renderer.domElement.addEventListener("pointerup", endPointer);
    renderer.domElement.addEventListener("pointercancel", endPointer);
    renderer.domElement.addEventListener("contextmenu", (event) => {
      event.preventDefault();
    });

    renderer.domElement.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        controls.radius = THREE.MathUtils.clamp(
          controls.radius * Math.exp(event.deltaY * 0.001),
          5.2,
          16
        );
        updateCamera();
      },
      { passive: false }
    );

    renderer.domElement.addEventListener("dblclick", resetView);
    document.getElementById("reset").addEventListener("click", resetView);

    window.addEventListener("keydown", (event) => {
      let handled = true;

      switch (event.key) {
        case "ArrowLeft":
          controls.theta += 0.08;
          break;
        case "ArrowRight":
          controls.theta -= 0.08;
          break;
        case "ArrowUp":
          controls.phi = Math.max(0.32, controls.phi - 0.06);
          break;
        case "ArrowDown":
          controls.phi = Math.min(1.48, controls.phi + 0.06);
          break;
        case "+":
        case "=":
          controls.radius = Math.max(5.2, controls.radius - 0.4);
          break;
        case "-":
        case "_":
          controls.radius = Math.min(16, controls.radius + 0.4);
          break;
        case "r":
        case "R":
          resetView();
          break;
        default:
          handled = false;
      }

      if (handled) {
        event.preventDefault();
        updateCamera();
      }
    });

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    updateCamera();
    renderer.setAnimationLoop(() => renderer.render(scene, camera));
  </script>

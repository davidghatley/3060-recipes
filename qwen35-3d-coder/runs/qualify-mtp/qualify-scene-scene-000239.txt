          .addScaledVector(right, -dx * panScale)
          .addScaledVector(up, dy * panScale);

    function animate() {
      const elapsed = clock.getElapsedTime();
      const delta = Math.min(clock.getDelta(), 0.05);

      const currentFlow = Number(flowInput.value);
      const dropSpeed = (currentFlow / 100) * 0.9 + 0.18;
      const columnOpacity = 0.12 + (currentFlow / 100) * 0.18;

      for (const emitter of emitterDrops) {
        emitter.column.material.opacity = columnOpacity;
        emitter.column.scale.y = 0.92 + Math.sin(elapsed * 2.1) * 0.04;

        for (const drop of emitter.drops) {
          const fall = (elapsed * drop.speed + drop.phase) % 1;
          drop.mesh.position.y = emitter.baseY - fall * emitter.fallDistance;
          drop.mesh.position.x =
            emitter.x + Math.sin(elapsed * 4 + drop.phase * 18) * 0.012;
          drop.mesh.scale.setScalar(0.75 + fall * 1.35);
        }
      }

      if (!reducedMotion) {
        cyanLight.intensity = 22 + Math.sin(elapsed * 1.7) * 3;
      }

      updateLabels();
      renderer.render(scene, camera);
    }

    renderer.setAnimationLoop(animate);

    requestAnimationFrame(() => {
      loading.classList.add("hidden");
      setTimeout(() => loading.remove(), 450);
    });
  </script>
</body>
</html>

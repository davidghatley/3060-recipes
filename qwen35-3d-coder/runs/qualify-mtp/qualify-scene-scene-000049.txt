  <script type="module">
    import * as THREE from
      "https://cdnjs.cloudflare.com/ajax/libs/three.js/0.160.0/three.module.min.js";

    const host = document.getElementById("scene");
    const loading = document.getElementById("loading");

    const scene = new THREE.Scene();
    scene.fog = new THREE.Fog(0x07111d, 9, 20);

    const camera = new THREE.PerspectiveCamera(
      35,
      window.innerWidth / window.innerHeight,
      0.1,
      100
    );

    const renderer = new THREE.WebGLRenderer({
      antialias: true,
      alpha: true,
      powerPreference: "high-performance"
    });

    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setClearColor(0x000000, 0);
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.12;
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;

    renderer.domElement.setAttribute(
      "aria-label",
      "Interactive three-dimensional ball-and-stick water molecule"
    );

    host.appendChild(renderer.domElement);

    // Lighting
    scene.add(new THREE.HemisphereLight(0xc7e9ff, 0x08111c, 1.65));

    const keyLight = new THREE.DirectionalLight(0xffffff, 3.2);
    keyLight.position.set(4.5, 6.5, 5.5);
    keyLight.castShadow = true;
    keyLight.shadow.mapSize.set(1024, 1024);
    keyLight.shadow.camera.left = -4;
    keyLight.shadow.camera.right = 4;
    keyLight.shadow.camera.top = 4;
    keyLight.shadow.camera.bottom = -4;
    keyLight.shadow.camera.near = 0.5;
    keyLight.shadow.camera.far = 15;
    keyLight.shadow.bias = -0.0002;
    scene.add(keyLight);

    const fillLight = new THREE.PointLight(0x55baff, 15, 12, 2);
    fillLight.position.set(-4, 2.5, 3.5);
    scene.add(fillLight);

    const rimLight = new THREE.PointLight(0xff6b78, 10, 10, 2);
    rimLight.position.set(3.5, 1, -2.5);
    scene.add(rimLight);

    // Molecule geometry
    const molecule = new THREE.Group();
    scene.add(molecule);

    const oxygenPosition = new THREE.Vector3(0, 2.15, 0);
    const bondAngle = THREE.MathUtils.degToRad(104.5);
    const halfBondAngle = bondAngle / 2;
    const bondLength = 1.45;

    const hydrogenOnePosition = new THREE.Vector3(
      -Math.sin(halfBondAngle) * bondLength,
      oxygenPosition.y - Math.cos(halfBondAngle) * bondLength,
      0
    );

    const hydrogenTwoPosition = new THREE.Vector3(
      Math.sin(halfBondAngle) * bondLength,
      oxygenPosition.y - Math.cos(halfBondAngle) * bondLength,
      0
    );

    const oxygenMaterial = new THREE.MeshPhysicalMaterial({
      color: 0xff3552,
      roughness: 0.2,
      metalness: 0.02,
      clearcoat: 0.7,
      clearcoatRoughness: 0.16,
      emissive: 0x260008,
      emissiveIntensity: 0.12
    });

    const hydrogenMaterial = new THREE.MeshPhysicalMaterial({
      color: 0xf8f8f8,
      roughness: 0.28,
      metalness: 0,
      clearcoat: 0.5,
      clearcoatRoughness: 0.18
    });

    const bondMaterial = new THREE.MeshStandardMaterial({
      color: 0x9ca8b0,
      roughness: 0.46,
      metalness: 0.28
    });

    const sleeveMaterial = new THREE.MeshStandardMaterial({
      color: 0xb9c2ca,
      roughness: 0.38,
      metalness: 0.42,
      side: THREE.DoubleSide
    });

    function addSphere(position, radius, material) {
      const geometry = new THREE.SphereGeometry(radius, 48, 32);
      const mesh = new THREE.Mesh(geometry, material);
      mesh.position.copy(position);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      molecule.add(mesh);
      return mesh;
    }

    addSphere(oxygenPosition, 0.58, oxygenMaterial);
    addSphere(hydrogenOnePosition, 0.38, hydrogenMaterial);
    addSphere(hydrogenTwoPosition, 0.38, hydrogenMaterial);

    function addBond(start, end) {
      const direction = new THREE.Vector3().subVectors(end, start);
      const length = direction.length();

      const geometry = new THREE.CylinderGeometry(
        0.105,
        0.105,
        length,
        24,
        1,
        false
      );

      const bond = new THREE.Mesh(geometry, bondMaterial);
      bond.position.copy(start).add(end).multiplyScalar(0.5);
      bond.quaternion.setFromUnitVectors(
        new THREE.Vector3(0, 1, 0),
        direction.clone().normalize()
      );
      bond.castShadow = true;
      bond.receiveShadow = true;
      molecule.add(bond);

      // Shallow, flared sleeves surrounding each atom.
      function addSleeve(position, axis, radius, length) {
        const group = new THREE.Group();

        const sleeveGeometry = new THREE.CylinderGeometry(
          radius * 1.18,
          radius * 0.55,
          length,
          32,
          1,
          true
        );

        const sleeve = new THREE.Mesh(sleeveGeometry, sleeveMaterial);
        sleeve.quaternion.setFromUnitVectors(
          new THREE.Vector3(0, 1, 0),
          axis.clone().normalize()
        );
        sleeve.castShadow = true;
        group.add(sleeve);

        for (const y of [-length / 2, length / 2]) {
          const ring = new THREE.Mesh(
            new THREE.TorusGeometry(radius * 1.1, 0.055, 10, 32),
            sleeveMaterial
          );

          ring.position.y = y;
          ring.quaternion.setFromUnitVectors(
            new THREE.Vector3(0, 0, 1),
            new THREE.Vector3(0, 1, 0)
          );
          ring.castShadow = true;
          group.add(ring);
        }

        group.position.copy(position);
        group.quaternion.setFromUnitVectors(
          new THREE.Vector3(0, 1, 0),
          axis.clone().normalize()
        );

        molecule.add(group);
      }

      const direction = new THREE.Vector3().subVectors(end, start).normalize();

      // Sleeve at the oxygen end.
      addSleeve(
        oxygenPosition.clone().addScaledVector(direction, 0.1),
        direction,
        0.69,
        0.28
      );

      // Sleeve at the hydrogen end.
      addSleeve(
        end.clone().addScaledVector(direction, -0.1),
        direction,
        0.49,
        0.24
      );
    }

    addBond(oxygenPosition, hydrogenOnePosition);
    addBond(oxygenPosition, hydrogenTwoPosition);

    // Direct camera controls, implemented without external controls modules.
    const controls = {
      theta: 0.32,
      phi: 1.37,
      radius: 5.8,
      target: new THREE.Vector3(0, 1.48, 0)
    };

    function updateCamera() {
      const sinPhi = Math.sin(controls.phi);
      const cosPhi = Math.cos(controls.phi);

      camera.position.set(
        controls.target.x +
          controls.radius * sinPhi * Math.sin(controls.theta),
        controls.target.y + controls.radius * cosPhi,
        controls.target.z +
          controls.radius * sinPhi * Math.cos(controls.theta)
      );

      camera.lookAt(controls.target);
    }

    function resetView() {
      controls.theta = 0.32;
      controls.phi = 1.37;
      controls.radius = 5.8;
      controls.target.set(0, 1.48, 0);
      updateCamera();
    }

    updateCamera();

    let dragging = false;
    let lastPointerX = 0;
    let lastPointerY = 0;

    renderer.domElement.addEventListener("pointerdown", (event) => {
      dragging = true;
      lastPointerX = event.clientX;
      lastPointerY = event.clientY;
      renderer.domElement.setPointerCapture(event.pointerId);
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - lastPointerX;
      const deltaY = event.clientY - lastPointerY;

      lastPointerX = event.clientX;
      lastPointerY = event.clientY;

      controls.theta -= deltaX * 0.006;
      controls.phi = THREE.MathUtils.clamp(
        controls.phi - deltaY * 0.005,
        0.35,
        Math.PI - 0.35
      );

      updateCamera();
    });

    function stopDragging(event) {
      dragging = false;

      if (
        event.pointerId !== undefined &&
        renderer.domElement.hasPointerCapture(event.pointerId)
      ) {
        renderer.domElement.releasePointerCapture(event.pointerId);
      }
    }

    renderer.domElement.addEventListener("pointerup", stopDragging);
    renderer.domElement.addEventListener("pointercancel", stopDragging);

    renderer.domElement.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();

        controls.radius = THREE.MathUtils.clamp(
          controls.radius * Math.exp(event.deltaY * 0.001),
          4.1,
          9.5
        );

        updateCamera();
      },
      { passive: false }
    );

    renderer.domElement.addEventListener("dblclick", resetView);
    renderer.domElement.addEventListener("contextmenu", (event) => {
      event.preventDefault();
    });

    document.getElementById("reset").addEventListener("click", resetView);

    // Projected atom labels
    const labels = [
      {
        element: document.getElementById("label-oxygen"),
        position: oxygenPosition.clone().add(new THREE.Vector3(0, 0.78, 0))
      },
      {
        element: document.getElementById("label-hydrogen-one"),
        position: hydrogenOnePosition.clone().add(
          new THREE.Vector3(
            -hydrogenOnePosition.x / Math.hypot(
              hydrogenOnePosition.x,
              hydrogenOnePosition.y - oxygenPosition.y
            ),
            0.48,
            0
          )
        ).add(new THREE.Vector3(0, 0.18, 0))
      },
      {
        element: document.getElementById("label-hydrogen-two"),
        position: hydrogenTwoPosition.clone().add(
          new THREE.Vector3(
            hydrogenTwoPosition.x / Math.hypot(
              hydrogenTwoPosition.x,
              hydrogenTwoPosition.y - oxygenPosition.y
            ),
            0.48,
            0
          )
        ).add(new THREE.Vector3(0, 0.18, 0))
      }
    ];

    function updateLabels() {
      const width = window.innerWidth;
      const height = window.innerHeight;

      for (const label of labels) {
        const projected = label.position.clone().project(camera);

        const visible = projected.z > -1 && projected.z < 1;
        const x = (projected.x * 0.5 + 0.5) * width;
        const y = (-projected.y * 0.5 + 0.5) * height;

        label.element.style.left = `${x}px`;
        label.element.style.top = `${y}px`;
        label.element.style.opacity = visible ? "1" : "0";
      }
    }

    function resize() {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();

      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    }

    window.addEventListener("resize", resize);

    renderer.setAnimationLoop(() => {
      renderer.render(scene, camera);
      updateLabels();
    });

    requestAnimationFrame(() => {
      loading.classList.add("hidden");
      setTimeout(() => loading.remove(), 400);
    });
  </script>
</body>

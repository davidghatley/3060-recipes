<body>
  <div id="app"></div>

  <header class="header">
    <div class="eyebrow">Molecular Structure</div>
    <h1>Water</h1>
    <div class="subtitle">A ball-and-stick representation of the H₂O molecule</div>
    <div class="badge">104° H–O–H bond angle</div>

  <script type="module">
    import * as THREE from
      "https://cdnjs.cloudflare.com/ajax/libs/three.js/0.160.0/three.module.min.js";

    const app = document.getElementById("app");

    // Renderer
    const renderer = new THREE.WebGLRenderer({
      antialias: true,
      alpha: true,
      powerPreference: "high-performance"
    });

    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.15;
    renderer.domElement.setAttribute(
      "aria-label",
      "Interactive three-dimensional water molecule"
    );
    app.appendChild(renderer.domElement);

    // Scene and camera
    const scene = new THREE.Scene();
    scene.fog = new THREE.Fog(0x08131f, 9, 18);

    const camera = new THREE.PerspectiveCamera(
      38,
      window.innerWidth / window.innerHeight,
      0.1,
      100
    );

    const target = new THREE.Vector3(0, 1.0, 0);

    const cameraState = {
      theta: 0.55,
      phi: 1.28,
      radius: 6.4
    };

    function updateCamera() {
      const { theta, phi, radius } = cameraState;

      camera.position.set(
        target.x + radius * Math.sin(phi) * Math.sin(theta),
        target.y + radius * Math.cos(phi),
        target.z + radius * Math.sin(phi) * Math.cos(theta)
      );

      camera.lookAt(target);
    }

    updateCamera();

    // Lighting
    scene.add(new THREE.HemisphereLight(0xc8e8ff, 0x07101a, 1.8));

    const keyLight = new THREE.DirectionalLight(0xffffff, 3.0);
    keyLight.position.set(4, 6, 5);
    keyLight.castShadow = true;
    keyLight.shadow.mapSize.set(1024, 1024);
    keyLight.shadow.camera.near = 0.5;
    keyLight.shadow.camera.far = 15;
    keyLight.shadow.bias = -0.0004;
    scene.add(keyLight);

    const rimLight = new THREE.DirectionalLight(0x4b8cff, 2.0);
    rimLight.position.set(-4, 2, -3);
    scene.add(rimLight);

    const oxygenGlow = new THREE.PointLight(0xff3055, 1.2, 7, 2);
    oxygenGlow.position.set(0, 1.6, 0);
    scene.add(oxygenGlow);

    // Molecule geometry
    const molecule = new THREE.Group();
    molecule.rotation.y = -0.18;
    scene.add(molecule);

    const oxygenPosition = new THREE.Vector3(0, 1.6, 0);

    // The two O-H bonds form a 104-degree H-O-H angle.
    const bondLength = 0.95;
    const halfBondAngle = THREE.MathUtils.degToRad(52);

    const hydrogenPositions = [-bondLength * Math.sin(halfBondAngle),
                                 bondLength * Math.sin(halfBondAngle)];

    const hydrogenHeights = oxygenPosition.y - bondLength * Math.cos(halfBondAngle);

    const oxygen = new THREE.Mesh(
      new THREE.SphereGeometry(0.36, 64, 48),
      new THREE.MeshPhysicalMaterial({
        color: 0xff3055,
        emissive: 0x3c0008,
        emissiveIntensity: 0.18,
        roughness: 0.2,
        metalness: 0,
        clearcoat: 1,
        clearcoatRoughness: 0.12
      })
    );

    oxygen.position.copy(oxygenPosition);
    molecule.add(oxygen);

    const hydrogenMaterial = new THREE.MeshPhysicalMaterial({
      color: 0xf4f9ff,
      emissive: 0x101722,
      emissiveIntensity: 0.12,
      roughness: 0.18,
      metalness: 0,
      clearcoat: 0.8,
      clearcoatRoughness: 0.15
    });

    const hydrogenGeometry = new THREE.SphereGeometry(0.235, 48, 36);

    hydrogenPositions.forEach((x) => {
      const hydrogen = new THREE.Mesh(hydrogenGeometry, hydrogenMaterial);
      hydrogen.position.set(x, hydrogenHeights, 0);
      molecule.add(hydrogen);
    });

    const bondMaterial = new THREE.MeshStandardMaterial({
      color: 0x9ba8b8,
      roughness: 0.48,
      metalness: 0.22
    });

    const sleeveMaterial = new THREE.MeshPhysicalMaterial({
      color: 0x8794a3,
      roughness: 0.32,
      metalness: 0.38,
      clearcoat: 0.5,
      side: THREE.DoubleSide
    });

    const ringMaterial = new THREE.MeshStandardMaterial({
      color: 0x657487,
      roughness: 0.3,
      metalness: 0.65
    });

    const yAxis = new THREE.Vector3(0, 1, 0);
    const zAxis = new THREE.Vector3(0, 0, 1);

    /**
     * Creates a cylinder between two points.
     */
    function addBond(a, b, radius) {
      const direction = new THREE.Vector3().subVectors(b, a);
      const length = direction.length();

      const geometry = new THREE.CylinderGeometry(
        radius,
        radius,
        length,
        24,
        1,
        false
      );

      const mesh = new THREE.Mesh(geometry, bondMaterial);
      mesh.position.copy(a).add(b).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(yAxis, direction.normalize());
      molecule.add(mesh);
    }

    /**
     * Creates a hollow, shallow sleeve around an atom.
     * The outer rim is placed on the atom side, and the opening faces the bond.
     */
    function addSleeve(center, outwardAxis, outerRadius, innerRadius, length) {
      // CylinderGeometry(radiusTop, radiusBottom, height).
      const geometry = new THREE.CylinderGeometry(
        innerRadius,
        outerRadius,
        length,
        32,
        1,
        true
      );

      const sleeve = new THREE.Mesh(geometry, sleeveMaterial);
      sleeve.position.copy(center);
      sleeve.quaternion.setFromUnitVectors(yAxis, outwardAxis.clone().normalize());
      molecule.add(sleeve);

      // Visible outer rim of the sleeve.
      const rim = new THREE.Mesh(
        new THREE.TorusGeometry(outerRadius * 0.94, 0.025, 12, 48),
        ringMaterial
      );

      rim.position.copy(center).addScaledVector(
        outwardAxis,
        Math.sign(outwardAxis.y) * length * 0.5
      );

      rim.quaternion.setFromUnitVectors(zAxis, outwardAxis.clone().normalize());
      molecule.add(rim);
    }

    const leftHydrogen = new THREE.Vector3(
      hydrogenPositions[0],
      hydrogenHeights,
      0
    );

    const rightHydrogen = new THREE.Vector3(
      hydrogenPositions[1],
      hydrogenHeights,
      0
    );

    // Main O-H bonds
    addBond(oxygenPosition, leftHydrogen, 0.068);
    addBond(oxygenPosition, rightHydrogen, 0.068);

    // Shallow sleeves at both ends of each bond
    addSleeve(
      oxygenPosition,
      new THREE.Vector3().subVectors(leftHydrogen, oxygenPosition),
      0.145,
      0.105,
      0.18
    );

    addSleeve(
      leftHydrogen,
      new THREE.Vector3().subVectors(oxygenPosition, leftHydrogen),
      0.125,
      0.09,
      0.15
    );

    addSleeve(
      oxygenPosition,
      new THREE.Vector3().subVectors(rightHydrogen, oxygenPosition),
      0.145,
      0.105,
      0.18
    );

    addSleeve(
      rightHydrogen,
      new THREE.Vector3().subVectors(oxygenPosition, rightHydrogen),
      0.125,
      0.09,
      0.15
    );

    // Subtle contact shadow plane
    const shadowPlane = new THREE.Mesh(
      new THREE.PlaneGeometry(20, 20),
      new THREE.ShadowMaterial({
        color: 0x000000,
        opacity: 0.18
      })
    );

    shadowPlane.rotation.x = -Math.PI / 2;
    shadowPlane.position.y = -0.72;
    shadowPlane.receiveShadow = true;
    scene.add(shadowPlane);

    molecule.traverse((object) => {
      if (object.isMesh) {
        object.castShadow = true;
        object.receiveShadow = true;
      }
    });

    // Projected atom labels
    const labels = [
      {
        element: document.getElementById("label-oxygen"),
        position: oxygenPosition
      },
      {
        element: document.getElementById("label-hydrogen-1"),
        position: leftHydrogen
      },
      {
        element: document.getElementById("label-hydrogen-2"),
        position: rightHydrogen
      }
    ];

    const projected = new THREE.Vector3();

    function updateLabels() {
      const width = window.innerWidth;
      const height = window.innerHeight;

      labels.forEach((label) => {
        projected.copy(label.position);
        molecule.localToWorld(projected);
        projected.project(camera);

        const visible =
          projected.z > -1 &&
          projected.z < 1 &&
          Math.abs(projected.x) < 1.15 &&
          Math.abs(projected.y) < 1.15;

        label.element.style.display = visible ? "block" : "none";

        if (visible) {
          const x = (projected.x * 0.5 + 0.5) * width;
          const y = (-projected.y * 0.5 + 0.5) * height;

          label.element.style.left = `${x}px`;
          label.element.style.top = `${y}px`;
        }
      });
    }

    // Direct camera controls: pointer drag, wheel, touch, keyboard
    let dragging = false;
    let lastX = 0;
    let lastY = 0;

    const canvas = renderer.domElement;

    canvas.addEventListener("pointerdown", (event) => {
      dragging = true;
      lastX = event.clientX;
      lastY = event.clientY;
      canvas.setPointerCapture(event.pointerId);
      canvas.classList.add("dragging");
    });

    canvas.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - lastX;
      const deltaY = event.clientY - lastY;

      lastX = event.clientX;
      lastY = event.clientY;

      cameraState.theta -= deltaX * 0.006;
      cameraState.phi = THREE.MathUtils.clamp(
        cameraState.phi - deltaY * 0.006,
        0.25,
        Math.PI - 0.25
      );

      updateCamera();
    });

    function endDrag(event) {
      dragging = false;
      canvas.classList.remove("dragging");

      if (event.pointerId !== undefined && canvas.hasPointerCapture(event.pointerId)) {
        canvas.releasePointerCapture(event.pointerId);
      }
    }

    canvas.addEventListener("pointerup", endDrag);
    canvas.addEventListener("pointercancel", endDrag);
    canvas.addEventListener("lostpointercapture", () => {
      dragging = false;
      canvas.classList.remove("dragging");
    });

    canvas.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        cameraState.radius = THREE.MathUtils.clamp(
          cameraState.radius * Math.exp(event.deltaY * 0.001),
          3.6,
          10
        );
        updateCamera();
      },
      { passive: false }
    );

    function resetView() {
      cameraState.theta = 0.55;
      cameraState.phi = 1.28;
      cameraState.radius = 6.4;
      updateCamera();
    }

    canvas.addEventListener("dblclick", resetView);
    document.getElementById("reset").addEventListener("click", resetView);

    window.addEventListener("keydown", (event) => {
      let handled = true;

      switch (event.key) {
        case "ArrowLeft":
          cameraState.theta -= 0.08;
          break;
        case "ArrowRight":
          cameraState.theta += 0.08;
          break;
        case "ArrowUp":
          cameraState.phi = THREE.MathUtils.clamp(
            cameraState.phi - 0.06,
            0.25,
            Math.PI - 0.25
          );
          break;
        case "ArrowDown":
          cameraState.phi = THREE.MathUtils.clamp(
            cameraState.phi + 0.06,
            0.25,
            Math.PI - 0.25
          );
          break;
        case "+":
        case "=":
          cameraState.radius = Math.max(3.6, cameraState.radius - 0.3);
          break;
        case "-":
          cameraState.radius = Math.min(10, cameraState.radius + 0.3);
          break;
        case "r":
        case "R":
          resetView();
          break;
        default:
          handled = false;
      }

      if (handled) {
        event.preventDefault();
        updateCamera();
      }
    });

    // Resize handling
    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    // Render loop
    renderer.setAnimationLoop(() => {
      updateLabels();
      renderer.render(scene, camera);
    });
  </script>

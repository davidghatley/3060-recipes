    // Four individually adjustable, threaded support struts.
    function createAdjustableStrut(top, bottom) {
      const group = new THREE.Group();
      scene.add(group);

      const direction = top.clone().sub(bottom);
      const totalLength = direction.length();
      direction.normalize();

      const lowerTop = bottom.clone().lerp(top, 0.58);
      const upperStart = bottom.clone().lerp(top, 0.48);

      cylinderBetween(
        group,
        bottom.clone().add(new THREE.Vector3(0, 0.025, 0)),
        lowerTop,
        0.075,
        aluminumMaterial
      );

      cylinderBetween(
        group,
        upperStart,
        top,
        0.055,
        steelMaterial
      );

      // Threaded section of the upper tube.
      const threadLength = totalLength * 0.44;
      const thread = new THREE.Mesh(
        new THREE.TubeGeometry(
          new HelixCurve(0.077, threadLength, 18),
          140,
          0.011,
          6,
          false
        ),
        darkMetalMaterial
      );
      thread.position.copy(lowerTop);
      thread.quaternion.setFromUnitVectors(Y_AXIS, direction);
      thread.castShadow = true;
      group.add(thread);

      // Collar at the telescoping joint.
      cylinderBetween(
        group,
        lowerTop.clone().addScaledVector(direction, -0.055),
        lowerTop.clone().addScaledVector(direction, 0.055),
        0.095,
        darkMetalMaterial
      );

      // Base plate and anchor bolt.

        new THREE.BoxGeometry(0.34, 0.055, 0.34),

    // Direct camera controls: orbit, pan, wheel zoom, and touch pinch.
    const pointers = new Map();
    let dragMode = "orbit";
    let lastX = 0;
    let lastY = 0;
    let lastPinchDistance = 0;
    let cameraTween = null;

    function panCamera(dx, dy) {
      camera.updateMatrixWorld();

      const right = new THREE.Vector3().setFromMatrixColumn(
        camera.matrixWorld,
        0
      );
      const up = new THREE.Vector3().setFromMatrixColumn(
        camera.matrixWorld,
        1
      );
      const scale = spherical.radius * 0.00145;

      target.addScaledVector(right, -dx * scale);
      target.addScaledVector(up, dy * scale);

      target.x = THREE.MathUtils.clamp(target.x, -4, 4);
      target.y = THREE.MathUtils.clamp(target.y, 0.6, 5.2);
      target.z = THREE.MathUtils.clamp(target.z, -3, 3);
    }

    function pointerDistance() {
      const values = [...pointers.values()];
      if (values.length < 2) return 0;
      return Math.hypot(
        values[0].x - values[1].x,
        values[0].y - values[1].y
      );
    }

    canvas.addEventListener("contextmenu", (event) => {
      event.preventDefault();
    });

    canvas.addEventListener("pointerdown", (event) => {
      cameraTween = null;
      canvas.setPointerCapture(event.pointerId);

      pointers.set(event.pointerId, {
        x: event.clientX,
        y: event.clientY
      });

      if (pointers.size === 1) {
        dragMode = event.button === 2 || event.shiftKey ? "pan" : "orbit";
        lastX = event.clientX;
        lastY = event.clientY;
      } else if (pointers.size === 2) {
        dragMode = "pinch";
        lastPinchDistance = pointerDistance();
      }
    });

    canvas.addEventListener("pointermove", (event) => {
      if (!pointers.has(event.pointerId)) return;

      pointers.set(event.pointerId, {
        x: event.clientX,
        y: event.clientY
      });

      if (pointers.size === 1) {
        const dx = event.clientX - lastX;
        const dy = event.clientY - lastY;
        lastX = event.clientX;
        lastY = event.clientY;

        if (dragMode === "pan") {
          panCamera(dx, dy);
        } else {
          spherical.theta -= dx * 0.006;
          spherical.phi = THREE.MathUtils.clamp(
            spherical.phi - dy * 0.005,
            0.18,
            Math.PI / 2 - 0.025
          );
        }
      } else if (pointers.size === 2) {
        const distance = pointerDistance();
        if (lastPinchDistance > 0 && distance > 0) {
          spherical.radius = THREE.MathUtils.clamp(
            spherical.radius * (lastPinchDistance / distance),
            4.5,
            22
          );
        }
        lastPinchDistance = distance;
      }
    });

    function releasePointer(event) {
      pointers.delete(event.pointerId);

      if (pointers.size === 1) {
        const remaining = [...pointers.values()][0];
        lastX = remaining.x;
        lastY = remaining.y;
        dragMode = "orbit";
      } else if (pointers.size === 0) {
        dragMode = "orbit";
      }
    }

    canvas.addEventListener("pointerup", releasePointer);
    canvas.addEventListener("pointercancel", releasePointer);

    canvas.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        cameraTween = null;
        spherical.radius = THREE.MathUtils.clamp(
          spherical.radius * Math.exp(event.deltaY * 0.001),
          4.5,
          22
        );
      },
      { passive: false }
    );

    function setView(position, lookTarget) {
      cameraTween = {
        start: performance.now(),
        fromSpherical: spherical.clone(),
        toSpherical: new THREE.Spherical().setFromVector3(
          position.clone().sub(lookTarget)
        ),
        fromTarget: target.clone(),
        toTarget: lookTarget.clone()
      };
    }

    window.addEventListener("keydown", (event) => {
      const key = event.key.toLowerCase();
      let handled = true;

      if (key === "arrowleft") spherical.theta += 0.08;
      else if (key === "arrowright") spherical.theta -= 0.08;
      else if (key === "arrowup") {
        spherical.phi = THREE.MathUtils.clamp(spherical.phi - 0.06, 0.18, 1.5);
      } else if (key === "arrowdown") {
        spherical.phi = THREE.MathUtils.clamp(spherical.phi + 0.06, 0.18, 1.5);
      } else if (key === "+" || key === "=") {
        spherical.radius = Math.max(4.5, spherical.radius * 0.9);
      } else if (key === "-") {
        spherical.radius = Math.min(22, spherical.radius * 1.1);
      } else if (key === "r") {
        setView(views.iso.position, views.iso.target);
      } else {
        handled = false;
      }

      if (handled) {
        event.preventDefault();
        cameraTween = null;
      }
    });

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    const clock = new THREE.Clock();
    let firstFrame = true;

    function animate(now) {
      requestAnimationFrame(animate);

      if (cameraTween) {
        const progress = Math.min(1, (now - cameraTween.start) / 1100);
        const eased = progress * progress * (3 - 2 * progress);

        spherical.copy(cameraTween.fromSpherical).lerp(
          cameraTween.toSpherical,
          eased
        );
        target.lerpVectors(
          cameraTween.fromTarget,
          cameraTween.toTarget,
          eased
        );

        const offset = new THREE.Vector3().setFromSpherical(spherical);
        camera.position.copy(target).add(offset);
        camera.lookAt(target);

        if (progress >= 1) cameraTween = null;
      }

      updateCamera();
      renderer.render(scene, camera);

      if (firstFrame) {
        firstFrame = false;
        loading.classList.add("hidden");
        setTimeout(() => loading.remove(), 450);
      }
    }

    requestAnimationFrame(animate);
  </script>

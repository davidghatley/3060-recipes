      dragging = true;
      lastX = event.clientX;
      lastY = event.clientY;
      canvas.setPointerCapture(event.pointerId);
      canvas.classList.add("dragging");
    });

    canvas.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - lastX;
      const deltaY = event.clientY - lastY;

      lastX = event.clientX;
      lastY = event.clientY;

      controls.theta -= deltaX * 0.006;
      controls.phi = THREE.MathUtils.clamp(
        controls.phi + deltaY * 0.005,
        0.35,
        1.48
      );

      updateCamera();
    });

    function endDrag(event) {
      dragging = false;
      canvas.classList.remove("dragging");

      if (event.pointerId !== undefined && canvas.hasPointerCapture(event.pointerId)) {
        canvas.releasePointerCapture(event.pointerId);
      }
    }

    canvas.addEventListener("pointerup", endDrag);
    canvas.addEventListener("pointercancel", endDrag);

    canvas.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        controls.radius = THREE.MathUtils.clamp(
          controls.radius * Math.exp(event.deltaY * 0.001),
          7,
          20
        );
        updateCamera();
      },
      { passive: false }
    );

    canvas.addEventListener("dblclick", resetView);

    window.addEventListener("keydown", (event) => {
      let handled = true;

      switch (event.key) {
        case "ArrowLeft":
          controls.theta += 0.08;
          break;
        case "ArrowRight":
          controls.theta -= 0.08;
          break;
        case "ArrowUp":
          controls.phi = THREE.MathUtils.clamp(controls.phi - 0.06, 0.35, 1.48);
          break;
        case "ArrowDown":
          controls.phi = THREE.MathUtils.clamp(controls.phi + 0.06, 0.35, 1.48);
          break;
        case "+":
        case "=":
          controls.radius = Math.max(7, controls.radius - 0.5);
          break;
        case "-":
          controls.radius = Math.min(20, controls.radius + 0.5);
          break;
        case "r":
        case "R":
          resetView();
          break;
        default:
          handled = false;
      }

      if (handled) {
        event.preventDefault();
        updateCamera();
      }
    });

    resetButton.addEventListener("click", resetView);

    orbitButton.addEventListener("click", () => {
      controls.auto = !controls.auto;
      orbitButton.setAttribute("aria-pressed", String(controls.auto));
      orbitButton.textContent = controls.auto ? "Stop orbit" : "Auto orbit";

    function addBamboo(start, end, radius = 0.075, material = bambooMaterial) {
      const direction = new THREE.Vector3().subVectors(end, start);
      const length = direction.length();

      const geometry = new THREE.CylinderGeometry(
        radius,
        radius,
        length,
        12
      );

      const mesh = new THREE.Mesh(geometry, material);
      mesh.position.copy(start).add(end).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(
        new THREE.Vector3(0, 1, 0),
        direction.normalize()
      );
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    function addBambooNode(position, direction, radius) {
      const geometry = new THREE.TorusGeometry(
        radius * 1.025,
        radius * 0.21,
        6,
        14
      );

      const node = new THREE.Mesh(geometry, bambooNodeMaterial);
      node.position.copy(position);
      node.quaternion.setFromUnitVectors(
        new THREE.Vector3(0, 0, 1),
        direction.clone().normalize()
      );
      node.castShadow = true;
      return node;
    }

    function addJoint(position, radius = 0.095) {
      const joint = new THREE.Mesh(
        new THREE.SphereGeometry(radius, 12, 9),
        bambooNodeMaterial
      );
      joint.position.copy(position);
      joint.castShadow = true;
      return joint;
    }

    function makeSaggingPanel(width, height, sag = 0.025) {
      const geometry = new THREE.PlaneGeometry(width, height, 14, 10);
      const positions = geometry.attributes.position;

      for (let i = 0; i < positions.count; i++) {
        const x = positions.getX(i);
        const y = positions.getY(i);

        const edgeX = Math.max(0, 1 - Math.pow(x / (width / 2), 2));
        const edgeY = Math.max(0, 1 - Math.pow(y / (height / 2), 2));

        positions.setZ(
          i,
          -sag * edgeX * edgeY
        );
      }

      geometry.computeVertexNormals();
      return geometry;
    }

    function addPanel(
      width,
      height,
      position,
      rotationY,
      material,
      sag = 0.025
    ) {
      const geometry = makeSaggingPanel(width, height, sag);
      const panel = new THREE.Mesh(geometry, material);
      panel.position.copy(position);
      panel.rotation.y = rotationY;
      panel.castShadow = true;
      panel.receiveShadow = true;
      return panel;
    }

    function addDashedLine(points, dashSize = 0.055, gapSize = 0.035) {
      const geometry = new THREE.BufferGeometry().setFromPoints(points);
      const material = new THREE.LineDashedMaterial({
        color: 0xf4c5a3,
        dashSize,
        gapSize,
        transparent: true,
        opacity: 0.62
      });

      const line = new THREE.Line(geometry, material);
      line.computeLineDistances();
      return line;
    }

    function makeTube(points, radius = 0.012) {
      const curve = new THREE.CatmullRomCurve3(points);
      const geometry = new THREE.TubeGeometry(
        curve,
        40,
        radius,
        5,
        false
      );

      const mesh = new THREE.Mesh(geometry, stringMaterial);
      mesh.castShadow = true;
      return mesh;
    }

    function createDiscGeometry() {
      const segments = 72;
      const rings = 12;
      const vertices = [0, 0, 0];
      const indices = [];

      for (let ring = 1; ring <= rings; ring++) {
        const fraction = ring / rings;

        for (let segment = 0; segment < segments; segment++) {
          const angle = (segment / segments) * Math.PI * 2;
          const irregularity =
            1 +
            0.035 * Math.sin(angle * 7 + ring * 1.4) +
            0.018 * Math.sin(angle * 13 - ring * 1.1);

          const radius = 0.72 * fraction * irregularity;

          vertices.push(
            Math.cos(angle) * radius,
            -0.15 * (1 - fraction * fraction) +
              0.008 * Math.sin(angle * 5 + ring) * fraction * fraction,
            Math.sin(angle) * radius
          );
        }
      }

      for (let segment = 0; segment < segments; segment++) {
        const next = (segment + 1) % segments;
        indices.push(0, 1 + next, 1 + segment);
      }

      for (let ring = 1; ring < rings; ring++) {
        const innerStart = 1 + (ring - 1) * segments;
        const outerStart = 1 + ring * segments;

        for (let segment = 0; segment < segments; segment++) {
          const next = (segment + 1) % segments;
          const a = innerStart + segment;
          const b = innerStart + next;
          const c = outerStart + segment;
          const d = outerStart + next;

          indices.push(a, b, c, a, c, d);
        }
      }

      const geometry = new THREE.BufferGeometry();
      geometry.setAttribute(
        "position",
        new THREE.Float32BufferAttribute(vertices, 3)
      );
      geometry.setIndex(indices);
      geometry.computeVertexNormals();
      return geometry;
    }

    function createArm(angle, lengthScale, phase) {
      const controlPoints = [
        { radius: 0.38, y: 4.68 },
        { radius: 0.9, y: 4.65 },
        { radius: 1.4, y: 4.45 },
        { radius: 1.85, y: 4.05 },
        { radius: 2.15, y: 3.35 },
        { radius: 2.35, y: 2.55 },
        { radius: 2.25, y: 1.85 },
        { radius: 1.9, y: 1.3 },
        { radius: 1.2, y: 1.05 },
        { radius: 0.65, y: 1.25 }
      ];

      const points = controlPoints.map((point, index) => {
        const t = index / (controlPoints.length - 1);
        const radial = point.radius * lengthScale;
        const y =
          4.68 +
          (point.y - 4.68) * lengthScale +
          Math.sin(t * 5 + phase) * 0.12 * t;

        return new THREE.Vector3(
          Math.cos(angle) * radial,
          y,
          Math.sin(angle) * radial
        );
      });

      const curve = new THREE.CatmullRomCurve3(
        points,
        false,
        "catmullrom",
        0.35
      );

      const segmentCount = 10;
      const startRadius = 0.145;
      const endRadius = 0.032;

      for (let i = 0; i < segmentCount; i++) {
        const t = i / (segmentCount - 1);
        const pointA = curve.getPoint(t);
        const pointB = curve.getPoint((i + 1) / segmentCount);

        const direction = new THREE.Vector3().subVectors(pointB, pointA);
        const length = direction.length();
        direction.normalize();

        const startRadiusValue = THREE.MathUtils.lerp(
          startRadius,
          endRadius,
          Math.pow(t, 0.82)
        );
        const endRadiusValue = THREE.MathUtils.lerp(
          startRadius * 0.84,
          endRadius,
          Math.pow(t, 0.82)
        );

        const geometry = new THREE.CylinderGeometry(
          endRadiusValue,
          startRadiusValue,
          length,
          14,
          1,
          false
        );

        const segment = new THREE.Mesh(geometry, armMaterial);
        segment.position.copy(pointA).add(pointB).multiplyScalar(0.5);
        segment.quaternion.setFromUnitVectors(up, direction);
        segment.castShadow = true;
        segment.receiveShadow = true;
        brittleStar.add(segment);
      }

      // Visible flexible joints between the long segments
      for (let i = 1; i < segmentCount; i++) {
        const t = i / segmentCount;
        const point = curve.getPoint(t);
        const jointRadius = THREE.MathUtils.lerp(0.115, 0.052, t);

        const joint = new THREE.Mesh(unitSphere, jointMaterial);
        joint.position.copy(point);
        joint.scale.setScalar(jointRadius);
        joint.castShadow = true;
        brittleStar.add(joint);
      }
    }

        target.y + radius * Math.cos(phi),
        target.z + radius * sinPhi * Math.cos(theta)

    }

    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - previousX;
      const deltaY = event.clientY - previousY;

      previousX = event.clientX;
      previousY = event.clientY;

      theta -= deltaX * 0.006;
      phi = THREE.MathUtils.clamp(phi - deltaY * 0.005, 0.45, 1.45);
      updateCamera();
    });

    function stopDragging(event) {
      dragging = false;
      if (event.pointerId !== undefined) {
        if (renderer.domElement.hasPointerCapture(event.pointerId)) {
          renderer.domElement.releasePointerCapture(event.pointerId);
        }
      }
    }

    renderer.domElement.addEventListener("pointerup", stopDragging);
    renderer.domElement.addEventListener("pointercancel", stopDragging);

    renderer.domElement.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        radius = THREE.MathUtils.clamp(
          radius + event.deltaY * 0.008,
          6.2,
          15
        );
        updateCamera();
      },
      { passive: false }
    );

    renderer.domElement.addEventListener("dblclick", resetCamera);
    updateCamera();

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);

      brittleStar.position.y = Math.sin(elapsed * 0.65) * 0.004;

      renderer.render(scene, camera);
    });

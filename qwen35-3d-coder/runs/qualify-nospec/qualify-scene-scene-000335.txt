          p.y += waveA + waveB + waveC;

          vec4 worldPosition = modelMatrix * vec4(p, 1.0);
          vWorldPosition = worldPosition.xyz;
          vWave = waveA + waveB + waveC;
          gl_Position = projectionMatrix * viewMatrix * worldPosition;
        }

      fragmentShader: `
        uniform float uTime;
        uniform vec3 uSunDirection;
        uniform vec3 uDeep;
        uniform vec3 uShallow;
        uniform vec3 uHorizon;
        varying vec3 vWorldPosition;
        varying float vWave;

        void main() {
          vec3 normal = normalize(vec3(
            -0.025 * cos(vWorldPosition.x * 0.18 + uTime * 0.72),
            1.0,
            -0.022 * cos(vWorldPosition.z * 0.23 - uTime * 0.54)
          ));

          vec3 viewDirection = normalize(cameraPosition - vWorldPosition);
          vec3 lightDirection = normalize(uSunDirection);

          float diffuse = max(dot(normal, lightDirection), 0.0);
          float specular = pow(
            max(dot(reflect(-lightDirection, normal), viewDirection), 0.0),
            110.0

            1.0 - max(dot(normal, viewDirection), 0.0),
            3.0
          );

          float distanceToCamera = length(cameraPosition.xz - vWorldPosition.xz);
          float haze = exp(-distanceToCamera * 0.011);

          vec3 color = mix(uDeep, uShallow, 0.35 + vWave * 1.4);
          color += vec3(0.38, 0.48, 0.43) * fresnel * 0.48;

          #include <tonemapping_fragment>
          #include <colorspace_fragment>
        }
      `

    const target = new THREE.Vector3(0, 1.25, 0);
    const spherical = new THREE.Spherical(16.5, 1.08, -2.28);
    const pointers = new Map();
    let autoOrbit = false;

    function updateCamera() {
      const offset = new THREE.Vector3().setFromSpherical(spherical);
      camera.position.copy(target).add(offset);
      camera.lookAt(target);
    }

    function resetView() {
      target.set(0, 1.25, 0);
      spherical.set(16.5, 1.08, -2.28);

    }

    renderer.domElement.addEventListener("contextmenu", (event) => {
      event.preventDefault();
    });

    renderer.domElement.addEventListener("pointerdown", (event) => {
      renderer.domElement.setPointerCapture(event.pointerId);
      pointers.set(event.pointerId, {
        x: event.clientX,
        y: event.clientY
      });
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      const previous = pointers.get(event.pointerId);
      if (!previous) return;

      const dx = event.clientX - previous.x;
      const dy = event.clientY - previous.y;

      previous.x = event.clientX;
      previous.y = event.clientY;

      if (event.button === 2) {
        const right = new THREE.Vector3().setFromMatrixColumn(camera.matrix, 0);
        const up = new THREE.Vector3().setFromMatrixColumn(camera.matrix, 1);
        const panScale = spherical.radius * 0.0013;

        target.addScaledVector(right, -dx * panScale);
        target.addScaledVector(up, dy * panScale);
      } else {
        spherical.theta -= dx * 0.005;
        spherical.phi = THREE.MathUtils.clamp(
          spherical.phi - dy * 0.005,
          0.22,
          1.47
        );
      }

      updateCamera();
    });

    function releasePointer(event) {
      pointers.delete(event.pointerId);
    }

    renderer.domElement.addEventListener("pointerup", releasePointer);
    renderer.domElement.addEventListener("pointercancel", releasePointer);

    renderer.domElement.addEventListener(
      "wheel",

        event.preventDefault();
        spherical.radius = THREE.MathUtils.clamp(
          spherical.radius * Math.exp(event.deltaY * 0.001),
          6.5,
          38
        );
        updateCamera();
      },
      { passive: false }
    );

    // ------------------------------------------------------------
    // Animation
    // ------------------------------------------------------------

    const clock = new THREE.Clock();
    let frame = 0;

    function animate() {
      const elapsed = clock.getElapsedTime();
      const delta = Math.min(clock.getDelta(), 0.05);

      waterMaterial.uniforms.uTime.value = elapsed;

      // Gentle boat motion.
      boat.position.y = Math.sin(elapsed * 0.72) * 0.065;
      boat.rotation.z = Math.sin(elapsed * 0.48) * 0.008;
      boat.rotation.x = Math.cos(elapsed * 0.43) * 0.005;

      // Crane motion and moving hook.
      slew.rotation.y = -0.12 + Math.sin(elapsed * 0.2) * 0.025;
      hookGroup.position.x = Math.sin(elapsed * 0.38) * 0.13;
      hookGroup.position.y = 1.72 + Math.sin(elapsed * 0.31) * 0.1;
      hookGroup.rotation.z = Math.sin(elapsed * 0.42) * 0.08;

      const hookTop = new THREE.Vector3(
        hookGroup.position.x,
        hookGroup.position.y + 0.2,
        hookGroup.position.z
      );
      setCylinderBetween(cable, boomTip, hookTop);

      radar.rotation.y = elapsed * 0.38;
      radar.rotation.z = Math.sin(elapsed * 0.6) * 0.08;

      // Navigation lights.
      const redPulse = Math.pow(Math.max(0, Math.sin(elapsed * 2.2)), 8);
      materials.redLight.emissiveIntensity = 0.8 + redPulse * 2.2;
      materials.greenLight.emissiveIntensity = 0.8 + redPulse * 2.2;

      // Buoy motion.
      buoy.position.y = Math.sin(elapsed * 0.85) * 0.08;
      buoy.rotation.z = Math.sin(elapsed * 0.62) * 0.035;
      buoy.rotation.x = Math.cos(elapsed * 0.55) * 0.025;
      buoyLight.rotation.y = elapsed * 1.7;

      // Wake animation.
      const wakeLeft = boat.children.find(
        (child) =>
          child.geometry?.type === "PlaneGeometry" &&
          child.position.x < 0
      );
      const wakeRight = boat.children.find(
        (child) =>
          child.geometry?.type === "PlaneGeometry" &&
          child.position.x > 0
      );

      if (wakeLeft && wakeRight) {
        wakeLeft.material.opacity = 0.13 + Math.sin(elapsed * 1.2) * 0.045;
        wakeRight.material.opacity = 0.13 + Math.sin(elapsed * 1.2 + 0.8) * 0.045;
        wakeLeft.position.x = -1.32 + Math.sin(elapsed * 0.45) * 0.08;
        wakeRight.position.x = 1.32 - Math.sin(elapsed * 0.45 + 0.5) * 0.08;
      }

      renderer.render(scene, camera);

      if (autoOrbit) {
        spherical.theta += delta * 0.08;
        updateCamera();
      }

      frame++;
      if (frame === 2) {
        document.getElementById("loading").classList.add("hidden");
      }
    }

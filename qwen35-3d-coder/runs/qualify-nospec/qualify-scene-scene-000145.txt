    function cylinderBetween(start, end, radius, material, radialSegments = 12) {
      const direction = new THREE.Vector3().subVectors(end, start);
      const length = direction.length();

      const geometry = new THREE.CylinderGeometry(
        radius,
        radius,
        length,
        radialSegments
      );

      mesh.quaternion.setFromUnitVectors(
        new THREE.Vector3(0, 1, 0),
        direction.normalize()
      );

      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    function makeStar(outerRadius, innerRadius, points) {
      const shape = new THREE.Shape();
      const radius = Math.PI * 2;

        const angle = (i / points) * radius;
        const r = i % 2 === 0 ? outerRadius : innerRadius;

        const x = Math.sin(angle) * r;
        const y = Math.cos(angle) * r;

        if (i === 0) {
          shape.moveTo(x, y);
        } else {
          shape.lineTo(x, y);
        }
      }

      shape.closePath();
      return shape;
    }

    let dragging = false;
    let previousX = 0;
    let previousY = 0;

    renderer.domElement.addEventListener("pointerdown", (event) => {
      dragging = true;
      previousX = event.clientX;
      previousY = event.clientY;
      lastInteraction = performance.now();
      renderer.domElement.setPointerCapture(event.pointerId);
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!dragging) return;

      const deltaX = event.clientX - previousX;
      const deltaY = event.clientY - previousY;

      previousX = event.clientX;
      previousY = event.clientY;

      cameraTheta -= deltaX * 0.005;
      cameraPhi = THREE.MathUtils.clamp(
        cameraPhi - deltaY * 0.0045,
        0.72,
        1.72
      );

      lastInteraction = performance.now();
      updateCamera();
    });

    function endDrag(event) {
      dragging = false;

      if (
        event.pointerId !== undefined &&
        renderer.domElement.hasPointerCapture(event.pointerId)
      ) {
        renderer.domElement.releasePointerCapture(event.pointerId);
      }
    }

    renderer.domElement.addEventListener("pointerup", endDrag);
    renderer.domElement.addEventListener("pointercancel", endDrag);

        updateCamera();

    let sigilActive = false;

    sigilButton.addEventListener("click", () => {
      sigilActive = !sigilActive;
      sigilButton.textContent = sigilActive
        ? "Sigil awakened"
        : "Awaken the sigil";

      if (sigilActive) {
        target.set(0, 0.2, 0);
        cameraRadius = 6.1;
        lastInteraction = performance.now();
      }
    });

    // -------------------------------------------------------------------------
    // Animation
    // -------------------------------------------------------------------------

    const clock = new THREE.Clock();
    const moteAttribute = moteGeometry.getAttribute("position");

    function animate() {
      const elapsed = clock.getElapsedTime();

      // Gentle automatic orbit after a short pause.
      if (
        !dragging &&
        performance.now() - lastInteraction > 3500
      ) {
        cameraTheta += 0.00055;
        updateCamera();
      }

      // Pulsing central sigil.
      const pulse = 1 + Math.sin(elapsed * 2.15) * 0.055;
      boss.scale.setScalar(pulse);

      materials.rune.opacity = 0.72 + Math.sin(elapsed * 2.15) * 0.2;
      materials.rune.color.setHSL(
        0.52,
        0.9,
        0.66 + Math.sin(elapsed * 2.15) * 0.08
      );

      bossLight.intensity = 3.2 + Math.sin(elapsed * 2.15) * 1.4;
      shieldLight.intensity = 8 + Math.sin(elapsed * 1.7) * 2;

      // Firefly drift.
      for (let i = 0; i < moteCount; i++) {
        const phase = motePhases[i];

        moteAttribute.array[i * 3] =
          moteBase[i * 3] + Math.sin(elapsed * 0.45 + phase) * 0.18;

        moteAttribute.array[i * 3 + 1] =
          moteBase[i * 3 + 1] + Math.sin(elapsed * 0.7 + phase) * 0.28;

        moteAttribute.array[i * 3 + 2] =
          moteBase[i * 3 + 2] + Math.cos(elapsed * 0.38 + phase) * 0.14;
      }

      moteAttribute.needsUpdate = true;

      // Subtle adventurer and shield breathing.
      adventurer.position.y = Math.sin(elapsed * 1.15) * 0.025;
      shield.position.y = Math.sin(elapsed * 1.15 + 0.7) * 0.012;

      renderer.render(scene, camera);
      requestAnimationFrame(animate);
    }

    animate();

    // -------------------------------------------------------------------------
    // Resize
    // -------------------------------------------------------------------------

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();

      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });
  </script>

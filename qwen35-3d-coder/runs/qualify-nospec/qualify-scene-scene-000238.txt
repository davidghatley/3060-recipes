      canvas.height = 256;

    function roundedRectShape(width, depth, radius) {
      const w = width / 2;
      const d = depth / 2;
      const r = Math.min(radius, w, d);
      const shape = new THREE.Shape();

      shape.moveTo(-w + r, -d);
      shape.lineTo(w - r, -d);
      shape.quadraticCurveTo(w, -d, w, -d + r);
      shape.lineTo(w, d - r);

      shape.lineTo(-w + r, d);
      shape.quadraticCurveTo(-w, d, -w, d - r);
      shape.lineTo(-w, -d + r);
      shape.quadraticCurveTo(-w, -d, -w + r, -d);
      return shape;

    function roundedPrismGeometry(width, depth, height, radius = 0.08) {
      const shape = roundedRectShape(width, depth, radius);
      const bevel = Math.min(0.025, height * 0.2);

        depth: height,
        steps: 1,
        curveSegments: 10,
        bevelEnabled: true,
        bevelSegments: 3,
        bevelSize: bevel,
        bevelThickness: bevel

      geometry.translate(0, height / 2, 0);
      geometry.computeVertexNormals();
      return geometry;

    // Creates a rail with genuine U-shaped notches cut into the top.
    function createNotchedRailGeometry(length, height, depth, notchCount) {
      const halfLength = length / 2;
      const halfDepth = depth / 2;
      const spacing = (length - depth) / notchCount;
      const shape = new THREE.Shape();

      shape.moveTo(-halfDepth, -height / 2);
      shape.lineTo(halfDepth, -height / 2);
      shape.lineTo(halfDepth, height / 2);

      // Traverse the top edge from right to left, forming each notch.
      for (let i = notchCount - 1; i >= 0; i--) {
        const centerX = (i - (notchCount - 1) / 2) * spacing;
        const notchRadius = depth / 2;

        shape.lineTo(centerX + notchRadius, height / 2);
        shape.lineTo(centerX, height / 2 - depth / 2);
        shape.quadraticCurveTo(
          centerX,
          height / 2 - depth,
          centerX + notchRadius,
          height / 2 - depth / 2
        );
        shape.lineTo(centerX - notchRadius, height / 2);
      }

      shape.lineTo(-halfDepth, height / 2);
      shape.closePath();

        depth: 0.12,
        steps: 1,
        bevelEnabled: true,
        bevelSegments: 2,
        bevelSize: 0.008,
        bevelThickness: 0.008

      geometry.translate(-0.06, 0, 0);
      geometry.computeVertexNormals();
      return geometry;

    const towel = new THREE.Mesh(
      roundedPrismGeometry(1.12, 0.76, 0.11, 0.065),
      new THREE.MeshStandardMaterial({
        color: 0x7899a5,
        roughness: 0.95,
        metalness: 0
      })
    );
    towel.position.set(0, 0.115, 0.83);
    towel.rotation.y = -0.025;
    towel.castShadow = true;
    towel.receiveShadow = true;
    shelfPivot.add(towel);

    const towelFold = new THREE.Mesh(
      roundedPrismGeometry(0.68, 0.49, 0.055, 0.05),
      new THREE.MeshStandardMaterial({
        color: 0xd7d2c4,
        roughness: 0.96,
        metalness: 0
      })
    );
    towelFold.position.set(-0.08, 0.195, 0.83);
    towelFold.rotation.y = 0.018;
    towelFold.castShadow = true;
    shelfPivot.add(towelFold);

    // Two hinge assemblies with small plates and visible pins.
    for (const side of [-1, 1]) {
      const x = side * 0.72;

      addBox(
        rack,
        new THREE.Vector3(0.045, 0.27, 0.32),
        new THREE.Vector3(x, 0.43, -0.72),
        metal
      );

      const pin = cylinderAlongX(0.043, 0.29, boltMaterial);
      pin.position.set(side * 0.76, 0.43, -0.72);
      rack.add(pin);

      for (const y of [0.355, 0.43, 0.505]) {
        const knuckle = cylinderAlongX(0.047, 0.19, metal);
        knuckle.position.set(side * 0.73, y, -0.72);
        rack.add(knuckle);
      }
    }

    // Custom orbit controls implemented directly, without Three.js addons.
    const pointers = new Map();
    let lastPinchDistance = 0;
    let dragging = false;
    let lastX = 0;
    let lastY = 0;

    function pointerDistance() {
      const values = [...pointers.values()];
      if (values.length < 2) return 0;
      return Math.hypot(
        values[0].x - values[1].x,
        values[0].y - values[1].y
      );
    }

    renderer.domElement.addEventListener("pointerdown", (event) => {
      renderer.domElement.setPointerCapture(event.pointerId);
      pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });

      if (pointers.size === 1) {
        dragging = true;
        lastX = event.clientX;
        lastY = event.clientY;
      } else if (pointers.size === 2) {
        dragging = false;
        lastPinchDistance = pointerDistance();
      }
    });

    renderer.domElement.addEventListener("pointermove", (event) => {
      if (!pointers.has(event.pointerId)) return;
      pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });

      if (pointers.size === 2) {
        const distance = pointerDistance();
        if (lastPinchDistance > 0 && distance > 0) {
          spherical.radius = THREE.MathUtils.clamp(
            spherical.radius * (lastPinchDistance / distance),
            3.4,
            9
          );
          updateCamera();
        }
        lastPinchDistance = distance;
        return;
      }

      if (!dragging) return;

      const dx = event.clientX - lastX;
      const dy = event.clientY - lastY;
      lastX = event.clientX;
      lastY = event.clientY;

      spherical.theta -= dx * 0.006;
      spherical.phi = THREE.MathUtils.clamp(
        spherical.phi - dy * 0.006,
        0.42,
        1.49
      );
      updateCamera();
    });

    function releasePointer(event) {
      pointers.delete(event.pointerId);
      if (pointers.size === 0) dragging = false;
      if (pointers.size < 2) lastPinchDistance = 0;
    }

    renderer.domElement.addEventListener("pointerup", releasePointer);
    renderer.domElement.addEventListener("pointercancel", releasePointer);
    renderer.domElement.addEventListener("lostpointercapture", releasePointer);

    renderer.domElement.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        spherical.radius = THREE.MathUtils.clamp(
          spherical.radius * Math.exp(event.deltaY * 0.001),
          3.4,
          9
        );
        updateCamera();
      },
      { passive: false }
    );

    renderer.domElement.addEventListener("dblclick", resetCamera);

    function resetCamera() {
      target.set(0, 1.45, 0);
      spherical.set(5.5, 1.37, 0.72);
      updateCamera();
    }

    // Shelf animation.
    const shelfButton = document.getElementById("shelfButton");
    const resetButton = document.getElementById("resetButton");

    let shelfOpen = true;
    let shelfTargetAngle = -0.38;
    let shelfAngle = shelfTargetAngle;

    shelfButton.addEventListener("click", () => {
      shelfOpen = !shelfOpen;
      shelfTargetAngle = shelfOpen ? -0.38 : 0;
      shelfButton.textContent = shelfOpen
        ? "Close lower shelf"
        : "Open lower shelf";
    });

    resetButton.addEventListener("click", resetCamera);

    window.addEventListener("keydown", (event) => {
      if (event.code === "Space") {
        event.preventDefault();
        shelfButton.click();
      }
      if (event.key.toLowerCase() === "r") resetCamera();
    });

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    renderer.setAnimationLoop(() => {
      shelfAngle = THREE.MathUtils.lerp(
        shelfAngle,
        shelfTargetAngle,
        0.085
      );
      shelfPivot.rotation.x = shelfAngle;
      renderer.render(scene, camera);
    });
  </script>
</body>

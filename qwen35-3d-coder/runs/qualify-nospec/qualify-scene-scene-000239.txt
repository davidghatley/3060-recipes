    );
    bed.receiveShadow = true;

    const soil = addMesh(
      new THREE.PlaneGeometry(12.25, 2.35),
      materials.soil,
      new THREE.Vector3(0, 0.005, 0),
      false
    );
    soil.rotation.x = -Math.PI / 2;

    // Main manifold.
    const manifoldY = 2.55;
    const manifoldZ = 0;
    const pipeStart = new THREE.Vector3(-6.25, manifoldY, manifoldZ);
    const pipeEnd = new THREE.Vector3(6.25, manifoldY, manifoldZ);

    cylinderBetween(pipeStart, pipeEnd, 0.14, materials.copper, 24);

    // Pipe end caps and flanges.
    for (const x of [-6.25, 6.25]) {
      const flange = addMesh(
        new THREE.CylinderGeometry(0.24, 0.24, 0.12, 24),
        materials.steel,
        new THREE.Vector3(x, manifoldY, manifoldZ)
      );
      flange.rotation.z = Math.PI / 2;

      const cap = addMesh(
        new THREE.CylinderGeometry(0.15, 0.15, 0.12, 20),
        materials.black,
        new THREE.Vector3(x + Math.sign(x) * 0.08, manifoldY, manifoldZ)
      );
      cap.rotation.z = Math.PI / 2;
    }

    // Eight emitter positions.
    const emitterPositions = [];
    for (let i = 0; i < 8; i++) {
      const x = (i - 3.5) * 1.3;
      emitterPositions.push(new THREE.Vector3(x, 0, manifoldZ));
    }

    const emitterData = emitterPositions.map((position, index) => {
      const branchTop = new THREE.Vector3(position.x, manifoldY, manifoldZ);
      const branchBottom = new THREE.Vector3(position.x, 0.48, manifoldZ);

      // Vertical supply line.
      cylinderBetween(branchTop, branchBottom, 0.075, materials.pipe, 14);

      // Supply connector.
      const connector = addMesh(
        new THREE.CylinderGeometry(0.12, 0.12, 0.18, 16),
        materials.steel,
        branchTop
      );
      connector.rotation.x = Math.PI / 2;

      // Filter housing.
      const filter = addMesh(
        new THREE.CylinderGeometry(0.16, 0.16, 0.27, 20),
        materials.black,
        new THREE.Vector3(position.x, 1.91, manifoldZ)
      );

      const filterCap = addMesh(
        new THREE.CylinderGeometry(0.18, 0.18, 0.055, 20),
        materials.steel,
        new THREE.Vector3(position.x, 2.065, manifoldZ)
      );

      // Valve body.
      const valve = addMesh(
        new THREE.CylinderGeometry(0.145, 0.145, 0.38, 18),
        materials.brass,
        new THREE.Vector3(position.x, 1.48, manifoldZ)
      );

      // Valve stem and handle.
      const stem = addMesh(
        new THREE.CylinderGeometry(0.035, 0.035, 0.34, 12),
        materials.steel,
        new THREE.Vector3(position.x, 1.72, manifoldZ)
      );

      const wheel = addMesh(
        new THREE.TorusGeometry(0.22, 0.032, 8, 28),
        materials.black,
        new THREE.Vector3(position.x, 1.91, manifoldZ)
      );
      wheel.rotation.x = Math.PI / 2;

      // Flexible tubing.
      const curve = new THREE.CatmullRomCurve3([
        new THREE.Vector3(position.x, 0.82, manifoldZ),
        new THREE.Vector3(position.x + 0.035, 0.56, manifoldZ + 0.025),
        new THREE.Vector3(position.x - 0.025, 0.29, manifoldZ),
        new THREE.Vector3(position.x, 0.2, manifoldZ)
      ]);

      const tubing = addMesh(
        new THREE.TubeGeometry(curve, 24, 0.065, 10, false),
        materials.rubber
      );

      // Emitter body and downward nozzle.
      const emitterBody = addMesh(
        new THREE.CylinderGeometry(0.12, 0.16, 0.25, 18),
        materials.brass,
        new THREE.Vector3(position.x, 0.27, manifoldZ)
      );

      const nozzle = addMesh(
        new THREE.CylinderGeometry(0.055, 0.085, 0.18, 16),
        materials.steel,
        new THREE.Vector3(position.x, 0.105, manifoldZ)
      );

      // Small status LED.
      const ledMaterial = new THREE.MeshStandardMaterial({
        color: 0x62ffca,
        emissive: 0x20d99d,
        emissiveIntensity: 2.2,
        roughness: 0.3
      });

      const led = addMesh(
        new THREE.SphereGeometry(0.045, 12, 8),
        ledMaterial,
        new THREE.Vector3(position.x, 1.31, manifoldZ + 0.12)
      );

      // Wet soil area.
      const wetSoil = addMesh(
        new THREE.CircleGeometry(0.58, 32),
        materials.wetSoil,
        new THREE.Vector3(position.x, 0.018, manifoldZ),
        false
      );
      wetSoil.rotation.x = -Math.PI / 2;

      // Tiny drainage lines in the soil.
      for (let j = 0; j < 3; j++) {
        const dropletLine = addMesh(
          new THREE.BufferGeometry().setFromPoints([
            new THREE.Vector3(position.x, 0.022, manifoldZ - 0.45 + j * 0.15),
            new THREE.Vector3(position.x + 0.34, 0.022, manifoldZ - 0.2 + j * 0.15)
          ]),
          new THREE.MeshBasicMaterial({
            color: 0x52695b,
            transparent: true,
            opacity: 0.22
          }),
          false
        );
        dropletLine.renderOrder = 1;
      }

      // Falling droplets.
      const droplets = [];
      for (let d = 0; d < 5; d++) {
        const droplet = addMesh(
          new THREE.SphereGeometry(0.035, 10, 8),
          materials.water.clone(),
          new THREE.Vector3(),
          false
        );
        droplet.userData.phase = d / 5;
        droplet.userData.offset = Math.random() * 0.18;
        droplets.push(droplet);
      }

      return {
        index,
        position,
        wheel,
        led,
        droplets,
        selected: false
      };
    });

    // Moving flow indicators along the manifold.
    const flowBeads = [];
    for (let i = 0; i < 12; i++) {
      const bead = addMesh(
        new THREE.SphereGeometry(0.075, 12, 8),
        materials.flow,
        new THREE.Vector3(),
        false
      );
      flowBeads.push(bead);
    }

    // Pressure gauge at the right end.
    const gaugeBody = addMesh(
      new THREE.CylinderGeometry(0.25, 0.25, 0.16, 28),
      materials.steel,
      new THREE.Vector3(6.48, manifoldY, 0.18)
    );
    gaugeBody.rotation.x = Math.PI / 2;

    const gaugeFace = addMesh(
      new THREE.CircleGeometry(0.215, 32),
      new THREE.MeshStandardMaterial({
        color: 0xe7eee8,
        roughness: 0.7
      }),
      new THREE.Vector3(6.57, manifoldY, 0.18),
      false
    );
    gaugeFace.rotation.y = Math.PI / 2;

    const gaugeNeedle = addMesh(
      new THREE.BoxGeometry(0.018, 0.15, 0.012),
      new THREE.MeshBasicMaterial({ color: 0xef5b45 }),
      new THREE.Vector3(6.59, manifoldY + 0.025, 0.205),
      false
    );
    gaugeNeedle.rotation.z = -0.65;

    const gaugeNumber = addMesh(
      new THREE.CircleGeometry(0.16, 24),
      new THREE.MeshBasicMaterial({
        color: 0x173338,
        transparent: true,
        opacity: 0.25,
        depthWrite: false
      }),
      new THREE.Vector3(6.57, manifoldY - 0.055, 0.185),
      false
    );
    gaugeNumber.rotation.y = Math.PI / 2;

    // Custom orbit controls.
    const target = new THREE.Vector3(0, 1.15, 0);
    const spherical = new THREE.Spherical(13.8, 1.15, 0.55);
    const initialSpherical = spherical.clone();
    const initialTarget = target.clone();

    let pointerDown = false;
    let pointerStartX = 0;
    let pointerStartY = 0;
    let pointerTravel = 0;

    function updateCamera() {
      spherical.phi = THREE.MathUtils.clamp(spherical.phi, 0.25, Math.PI * 0.49);
      spherical.radius = THREE.MathUtils.clamp(spherical.radius, 7, 24);
      camera.position.setFromSpherical(spherical).add(target);
      camera.lookAt(target);
    }

    updateCamera();

    canvas.addEventListener("pointerdown", (event) => {
      pointerDown = true;
      pointerTravel = 0;
      pointerStartX = event.clientX;
      pointerStartY = event.clientY;
      canvas.setPointerCapture(event.pointerId);
    });

    canvas.addEventListener("pointermove", (event) => {
      if (!pointerDown) return;

      const dx = event.clientX - pointerStartX;
      const dy = event.clientY - pointerStartY;
      pointerTravel += Math.abs(dx) + Math.abs(dy);

      spherical.theta -= dx * 0.006;
      spherical.phi -= dy * 0.005;
      updateCamera();
    });

    canvas.addEventListener("pointerup", (event) => {
      pointerDown = false;
      if (canvas.hasPointerCapture(event.pointerId)) {
        canvas.releasePointerCapture(event.pointerId);
      }

      if (pointerTravel < 6) {
        inspectEmitter(event.clientX, event.clientY);
      }
    });

    canvas.addEventListener("pointercancel", () => {
      pointerDown = false;
    });

    canvas.addEventListener(
      "wheel",
      (event) => {
        event.preventDefault();
        spherical.radius *= Math.exp(event.deltaY * 0.001);
        updateCamera();
      },
      { passive: false }
    );

    // Click inspection.
    const raycaster = new THREE.Raycaster();
    const pointer = new THREE.Vector2();
    let selectedEmitter = null;

    function inspectEmitter(clientX, clientY) {
      const rect = canvas.getBoundingClientRect();
      pointer.x = ((clientX - rect.left) / rect.width) * 2 - 1;
      pointer.y = -((clientY - rect.top) / rect.height) * 2 + 1;

      raycaster.setFromCamera(pointer, camera);
      const hits = raycaster.intersectObjects(emitterData, true);

      if (!hits.length) return;

      let inspected = emitterData[hits[0].object.userData.emitterIndex];
      if (!inspected) return;

      selectedEmitter = inspected;

      emitterData.forEach((emitter) => {
        emitter.selected = emitter === inspected;
      });

      selectionRing.style.left = `${clientX}px`;
      selectionRing.style.top = `${clientY}px`;
      selectionRing.style.opacity = "1";

      if (selectedEmitter) {
        document.querySelector(".subtitle").textContent =
          `Emitter ${String(selectedEmitter.index + 1).padStart(2, "0")}: ` +
          `${selectedEmitter.selected ? "Selected" : "Unselected"} · ` +
          `${selectedEmitter.selected ? "Flow active" : "Flow paused"}.`;
      }
    }

    // UI controls.
    let waterOn = true;
    let labelsVisible = true;

    waterButton.addEventListener("click", () => {
      waterOn = !waterOn;
      waterButton.textContent = waterOn ? "Water on" : "Water off";
      waterButton.classList.toggle("active", waterOn);
      statusDot.classList.toggle("off", !waterOn);
      statusText.textContent = waterOn ? "Watering active" : "System paused";
    });

    labelButton.addEventListener("click", () => {
      labelsVisible = !labelsVisible;
      labelButton.classList.toggle("active", labelsVisible);
      labelLayer.classList.toggle("hidden", !labelsVisible);
      selectionRing.style.opacity = "0";
    });

    resetButton.addEventListener("click", () => {
      spherical.copy(initialSpherical);
      target.copy(initialTarget);
      updateCamera();
      selectedEmitter = null;
      selectionRing.style.opacity = "0";
      emitterData.forEach((emitter) => {
        emitter.selected = false;
      });
      document.querySelector(".subtitle").textContent =
        "Select an emitter to inspect it. Drag to orbit, scroll to zoom, " +
        "and use the controls to manage the irrigation system.";
    });

    // Projected HTML labels.
    const labelElements = emitterData.map(() => {
      const element = document.createElement("div");
      element.className = "label";
      element.innerHTML = `<span class="number">${String(emitterData.indexOf(emitter) + 1).padStart(2, "0")}</span>`;
      labelLayer.appendChild(element);
      return element;
    });

    const projected = new THREE.Vector3();

    function updateLabels() {
      if (!labelsVisible) return;

      emitterData.forEach((emitter, index) => {
        projected.copy(emitter.position).setY(1.2).setZ(0.45).project(camera);

        const visible =
          projected.z > -1 &&
          projected.z < 1 &&
          Math.abs(projected.x) < 1.15 &&
          Math.abs(projected.y) < 1.15;

        labelElements[index].classList.toggle("hidden", !visible);

        if (visible) {
          const x = (projected.x * 0.5 + 0.5) * window.innerWidth;
          const y = (-projected.y * 0.5 + 0.5) * window.innerHeight;
          labelElements[index].style.left = `${x}px`;
          labelElements[index].style.top = `${y}px`;
        }
      });
    }

    function updateSelectionRing() {
      if (!selectedEmitter) {
        selectionRing.style.opacity = "0";
        return;
      }

      selectionRing.style.left = `${selectedEmitter.position.x}px`;
      selectionRing.style.top = `${0.28}px`;
      selectionRing.style.opacity = "1";
    }

    // Animation.
    const clock = new THREE.Clock();

    function animate() {
      requestAnimationFrame(animate);
      const elapsed = clock.getElapsedTime();

      emitterData.forEach((emitter) => {
        emitter.droplets.forEach((droplet) => {
          const progress = (elapsed * 0.72 + droplet.userData.phase) % 1;
          const fall = progress * 0.115;

          droplet.position.set(
            emitter.position.x + Math.sin(elapsed * 4 + droplet.userData.phase * 8) * 0.012,
            0.105 - fall,
            manifoldZ
          );

          const scale = 0.65 + Math.sin(progress * Math.PI) * 0.85;
          droplet.scale.setScalar(scale);
          droplet.material.opacity = waterOn ? 0.62 * (1 - progress * 0.45) : 0;
          droplet.visible = waterOn && progress < 0.98;
        });

        emitter.led.material.emissiveIntensity = waterOn ? 2.1 : 0.05;
        emitter.wheel.rotation.z = Math.sin(elapsed * 0.7 + emitter.index) * 0.08;
      });

      if (waterOn) {
        flowBeads.forEach((bead, index) => {
          const x = -5.8 + ((elapsed * 1.05 + index * 0.95) % 11.6);
          bead.position.set(x, manifoldY + 0.19, manifoldZ + 0.04);
          bead.scale.setScalar(0.85 + Math.sin(elapsed * 5 + index) * 0.15);
          bead.visible = true;
        });
      } else {
        flowBeads.forEach((bead) => {
          bead.visible = false;
        });
      }

      updateLabels();
      updateSelectionRing();

      renderer.render(scene, camera);
    }

    animate();

    window.addEventListener("resize", () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });
  </script>
</body>
</html>
